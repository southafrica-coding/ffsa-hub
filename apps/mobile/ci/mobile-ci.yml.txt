name: Mobile CI

# Checks for the phone app in apps/mobile. Independent of the website: it
# installs only apps/mobile's own dependencies. src/types/database.ts is
# included so a web schema change flags a stale mobile types copy.
on:
  pull_request:
    paths:
      - "apps/mobile/**"
      - "src/types/database.ts"
      - ".github/workflows/mobile-ci.yml"
  push:
    branches: [main]
    paths:
      - "apps/mobile/**"
      - "src/types/database.ts"
      - ".github/workflows/mobile-ci.yml"

permissions:
  contents: read

concurrency:
  group: mobile-ci-${{ github.ref }}
  cancel-in-progress: true

jobs:
  check:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    defaults:
      run:
        working-directory: apps/mobile
    steps:
      - uses: actions/checkout@v7

      - uses: actions/setup-node@v7
        with:
          # Expo SDK 58 supports Node ^22.13 || ^24.3 || >=26.
          node-version: 24
          cache: npm
          cache-dependency-path: apps/mobile/package-lock.json

      - run: npm ci
      - run: npm run types:check
      - run: npm run typecheck
      - run: npm run lint
      - run: npm test -- --ci
